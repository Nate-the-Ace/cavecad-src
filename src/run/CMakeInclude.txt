cmake_minimum_required(VERSION 3.16)
project(${PROJECT_NAME} VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Core Widgets Concurrent Gui Network OpenGL Sql Svg Xml PrintSupport Qml)
qt_standard_project_setup()

# CaveCAD's own version (the VERSION file), for the .exe's file properties
# and the macOS bundle -- not the QCAD base version in RVersion.h, which
# scripts compare against and which stays as it is.
file(STRINGS ${CMAKE_CURRENT_LIST_DIR}/../../VERSION CAVECAD_VERSION LIMIT_COUNT 1)
string(STRIP "${CAVECAD_VERSION}" CAVECAD_VERSION)
string(REPLACE "." ";" CAVECAD_VERSION_PARTS "${CAVECAD_VERSION}")
list(APPEND CAVECAD_VERSION_PARTS 0 0 0 0)
list(GET CAVECAD_VERSION_PARTS 0 CAVECAD_VERSION_MAJOR)
list(GET CAVECAD_VERSION_PARTS 1 CAVECAD_VERSION_MINOR)
list(GET CAVECAD_VERSION_PARTS 2 CAVECAD_VERSION_REV)
list(GET CAVECAD_VERSION_PARTS 3 CAVECAD_VERSION_BUILD)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_LIST_DIR}/../../VERSION)

if(WIN32)
    set(RC cavecad.rc)
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/cavecad_version.h
        "#define CAVECAD_VERSION_MAJOR ${CAVECAD_VERSION_MAJOR}\n"
        "#define CAVECAD_VERSION_MINOR ${CAVECAD_VERSION_MINOR}\n"
        "#define CAVECAD_VERSION_REV ${CAVECAD_VERSION_REV}\n"
        "#define CAVECAD_VERSION_BUILD ${CAVECAD_VERSION_BUILD}\n"
        "#define CAVECAD_VERSION_STRING \"${CAVECAD_VERSION}\"\n")
endif()

set(app_icon_macos "${CMAKE_CURRENT_LIST_DIR}/cavecad.icns")
set_source_files_properties(${app_icon_macos} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")

if(CMAKE_BUILD_TYPE STREQUAL "Release")
    link_directories(${CMAKE_CURRENT_LIST_DIR}/../../release)
else()
    link_directories(${CMAKE_CURRENT_LIST_DIR}/../../debug)
endif()

qt_add_executable(${PROJECT_NAME} WIN32 MACOSX_BUNDLE
    ${CMAKE_CURRENT_LIST_DIR}/main.cpp ${CMAKE_CURRENT_LIST_DIR}/${RC} ${app_icon_macos}
)

# CMake's default Info.plist otherwise says com.yourcompany, version 1.0
# and no icon at all.
set_target_properties(${PROJECT_NAME} PROPERTIES
    MACOSX_BUNDLE_ICON_FILE cavecad.icns
    MACOSX_BUNDLE_GUI_IDENTIFIER org.cavecad.CaveCAD
    MACOSX_BUNDLE_BUNDLE_NAME CaveCAD
    MACOSX_BUNDLE_BUNDLE_VERSION ${CAVECAD_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${CAVECAD_VERSION}
    MACOSX_BUNDLE_COPYRIGHT "GNU GPL v3. Based on QCAD, (c) RibbonSoft GmbH."
)

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}
    ${CMAKE_CURRENT_LIST_DIR}/../../src/3rdparty
    ${CMAKE_CURRENT_LIST_DIR}/../../src/core
    ${CMAKE_CURRENT_LIST_DIR}/../../src/core/math
    ${CMAKE_CURRENT_LIST_DIR}/../../src/entity
    ${CMAKE_CURRENT_LIST_DIR}/../../src/grid
    ${CMAKE_CURRENT_LIST_DIR}/../../src/gui
    ${CMAKE_CURRENT_LIST_DIR}/../../src/io/dxf
    ${CMAKE_CURRENT_LIST_DIR}/../../src/operations
    ${CMAKE_CURRENT_LIST_DIR}/../../src/scripting
    ${CMAKE_CURRENT_LIST_DIR}/../../src/snap
    ${CMAKE_CURRENT_LIST_DIR}/../../src/spatialindex
    ${CMAKE_CURRENT_LIST_DIR}/../../src/stemmer
    ${CMAKE_CURRENT_LIST_DIR}/../../src/zip
)

target_compile_definitions(${PROJECT_NAME} PRIVATE
    QCAD_DLL
)

if(CMAKE_BUILD_TYPE STREQUAL "Release")
    set_target_properties(${PROJECT_NAME} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}/../../release)
else()
    set_target_properties(${PROJECT_NAME} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}/../../debug)
endif()

if(${CMAKE_SYSTEM_NAME} STREQUAL "Linux")
    set_target_properties(${PROJECT_NAME} PROPERTIES OUTPUT_NAME cavecad-bin)
endif()
